from dataclasses import dataclass

from before import Summary


@dataclass(frozen=True)
class SummaryContract:
    truncated_reasons: frozenset[str]
    refused_reasons: frozenset[str]


class SilentFailureError(Exception):
    """契約に違反した応答を表す基底例外。"""


class EmptyResponseError(SilentFailureError):
    """本文が空だった。"""


class TruncatedResponseError(SilentFailureError):
    """打ち切られた。finish_reason が上限到達を示す場合と、
    応答そのものが欠落していた場合の両方でこの例外になる。"""


class RefusedResponseError(SilentFailureError):
    """安全性分類器の判断により拒否された。"""

def judge(reply: dict, contract: SummaryContract) -> None:
    try:
        finish_reason = reply["finish_reason"]
        body = reply["body"]
    except KeyError as e:
        raise TruncatedResponseError("応答が途中で欠落しています") from e

    if finish_reason in contract.truncated_reasons:
        raise TruncatedResponseError(finish_reason)
    if finish_reason in contract.refused_reasons:
        raise RefusedResponseError(finish_reason)
    if not body.strip():
        raise EmptyResponseError()
