// 言葉を含む出どころの名前を、sources に並んだ順で返す
// ponytail: 字面の一致だけ。言い換えられた根拠は拾えない。拾いたくなったら人が読む
export function whereFrom(phrase: string, sources: Record<string, string>): string[] {
  return Object.entries(sources)
    .filter(([, text]) => text.includes(phrase))
    .map(([name]) => name);

Record<string, string> は、キーも値も文字列のオブジェクトの型だ。ここでは、出どころの名前をキーに、その中身の文章を値にする。Object.entries は、オブジェクトを、キーと値の組の配列にする。includes は、その文字列が含まれているかを調べる。filter で、言葉を含む組だけを残して、map で名前だけを取り出す。

export const CRITERIA = [
  'あなたは、この差分を書いていない立場のレビュー役です。',
  '仕様の各要件が、差分で実装されているかを確かめてください。',
  '報告するのは、正しさと仕様に関わる食い違いだけです。スタイルや好みは報告しないでください。',
  '書いた側の説明は渡していません。差分と仕様に書いてあることだけで判断してください。',
].join('\n');

// 依頼は、基準文・仕様・差分の三つだけで組む。会話も委任文も、引数に取らない
// ponytail: 差分や仕様の中に ``` の行があると囲みが崩れる。出会ったら囲みの長さを変える
export function buildReviewRequest(diff: string, spec: string): string {
  return [
    CRITERIA,
    '',
    '## 仕様（main の SPEC.md）',
    '',
    '```markdown',
    spec.trimEnd(),
    '```',
    '',
    '## 差分（git diff main...HEAD）',
    '',
    '```diff',
    diff.trimEnd(),
    '```',
    '',
  ].join('\n');
}
