// 会話の記録（.jsonl）から Bash の command を取り出す。
// 書きかけの最後の行など、読めない行は飛ばす
export function extractCommands(jsonl: string): string[] {
  const commands: string[] = [];
  for (const line of jsonl.split('\n')) {
    let entry: any;
    try {
      entry = JSON.parse(line);
    } catch {
      continue;
    }
    if (entry?.type !== 'assistant' || !Array.isArray(entry.message?.content)) continue;
    for (const block of entry.message.content) {
      if (block?.type === 'tool_use' && block.name === 'Bash' && typeof block.input?.command === 'string') {
        commands.push(block.input.command);
      }
    }
  }
  return commands;
}

ファイルの中身を改行で一行ずつに分けて、JSON.parse で JavaScript の値に直す。JSON.parse は、JSON の文字列を読んで、オブジェクトや配列に直す関数だ。読めない文字列を渡すと、例外を投げる。Claude Code が書いている途中のファイルを読むと、最後の行が書きかけのことがあるので、try と catch で受けて、その行は飛ばす。entry に付けた any は、TypeScript に「この値の形は問わない」と伝える型だ。記録の形は決まった仕様ではないので、?. で一段ずつ、そのキーが無くてもエラーにならずに undefined になるようにたどっている。Array.isArray は、値が配列かどうかを確かめる。

// 回数の多い順。同数は文字列の順
export function countByFrequency(commands: string[]): [string, number][] {
  const counts = new Map<string, number>();
  for (const c of commands) counts.set(c, (counts.get(c) ?? 0) + 1);
  return [...counts].sort((a, b) => b[1] - a[1] || (a[0] < b[0] ? -1 : 1));
}

Map は、キーと値の組を覚えておく入れ物だ。コマンドをキーにして、回数を値にする。counts.get(c) は、まだ一度も出ていないコマンドだと undefined を返すので、?? 0 でそのときは0として扱う。[...counts] で、Map の中身を [コマンド, 回数] の組の配列に開く。戻り値の型の [string, number][] は、文字列と数の組が並んだ配列、という意味だ。sort に渡した関数は、二つの組を比べて、負なら a を前、正なら b を前に置く。b[1] - a[1] で回数の多いほうを前にし、回数が同じで0になったときだけ、|| の右側で、文字列の順に並べる。

// 許可設定と判定器を受け取り、コマンドの列に当てて数える
export function buildReplay(rules: Rules, judge: Judge): (commands: string[]) => Tally {
  return (commands) => {
    const tally: Tally = { allow: 0, ask: 0, deny: 0, asked: [], denied: [] };
    for (const command of commands) {
      const d = judge(rules, command);
      tally[d]++;
      const seen = d === 'ask' ? tally.asked : d === 'deny' ? tally.denied : null;
      if (seen !== null && !seen.includes(command)) seen.push(command);
    }
    return tally;

import type { Decision, Rules } from './types.ts';

// "Bash(...)" の中身を取り出して、コマンドに一致するかを見る
export function matchRule(rule: string, command: string): boolean {
  const m = /^Bash\((.*)\)$/s.exec(rule);
  if (m === null) return false;
  const pattern = m[1]!;
  if (pattern.endsWith(' *') || pattern.endsWith(':*')) {
    const head = pattern.slice(0, -2);
    return command === head || command.startsWith(head + ' ');
  }
  if (pattern.endsWith('*')) return command.startsWith(pattern.slice(0, -1));
  return command === pattern;
}

正規表現の ^Bash\((.*)\)$ で、Bash( と ) に挟まれた中身を取り出す。\( は、括弧そのものに一致させる書き方だ（ただの ( は、取り出す範囲の印になる）。最後の s は、. が改行にも一致するようにする印だ。exec は、一致しなければ null を、一致すれば一致した部分を並べた配列を返す。取り出した中身は m[1] に入る。その後ろの ! は、TypeScript に「ここは undefined ではない」と伝える印だ。一致したなら、中身は必ずある。

中身の末尾が * か :* なら、その手前（slice(0, -2) で末尾の2文字を落とした残り）とまったく同じか、手前に空白を足したもので始まれば一致にする。空白を足すのが、lsof を ls に一致させないための一手だ。空白の無い * で終わっていれば、手前の文字で始まるだけで一致にする。* が無ければ、完全に同じときだけ。

// ponytail: 完全なシェルの字句解析ではない。heredoc・バックスラッシュ・2>&1 の扱いは道具より粗い。
export function splitCommand(command: string): string[] | null {
  const parts: string[] = [];
  let quote: string | null = null;
  let start = 0;
  for (let i = 0; i < command.length; i++) {
    const c = command[i]!;
    const two = command.slice(i, i + 2);
    if (c === '`' || two === '$(') return null;
    if (quote !== null) {
      if (c === quote) quote = null;
      continue;
    }
    if (c === "'" || c === '"') quote = c;
    else if (two === '<(') return null;
    else if ('&|;\n'.includes(c)) {
      parts.push(command.slice(start, i));
      if (two === '&&' || two === '||') i++;
      start = i + 1;
    }
  }
  parts.push(command.slice(start));
  if (/(&&|\|\|)\s*$/.test(command)) return null;
  return parts.map((p) => p.trim()).filter((p) => p !== '');

// 評価順は deny、ask、allow。書いていないものは聞く
export function decide(rules: Rules, command: string): Decision {
  const parts = splitCommand(command);
  const hit = (list: string[], text: string) => list.some((rule) => matchRule(rule, text));
  if (parts === null) return hit(rules.deny, command) ? 'deny' : 'ask';
  if (parts.some((p) => hit(rules.deny, p))) return 'deny';
  if (parts.some((p) => hit(rules.ask, p))) return 'ask';
  return parts.every((p) => hit(rules.allow, p)) ? 'allow' : 'ask';
}

hit は、一覧のどれか一つのルールが一致するかを見る、小さな関数だ。some は、配列のどれか一つでも条件を満たせば真になる。every は、全部が条件を満たしたときだけ真になる。

分けられなかったコマンドは、deny にだけは全体の文字列で当てて、一致すれば止める。それ以外は、確認に回す。分けられたら、どれか一つでも deny に一致すれば止める。次に、どれか一つでも ask に一致すれば聞く。最後に、全部がそれぞれ allow に一致したときだけ通す。一つでも、どこにも一致しないものがあれば、聞く。書いていないものは、聞く。

夕方、帰りの保安ゲートで、臨時の通行証を返した。来たときに鷹宮さんの区域を読み上げていた警備員が、受け取った通行証と引き換えに、カードを一枚、窓口から差し出した。

「鷹宮さんから。付き添いの方に、って」

名刺より少し大きい、厚手のカードだった。手書きで、四行。一行ずつ、短かった。五行目の欄だけが、白いまま残っていた。

帰りの作業車の助手席で、俺はそのカードを膝の上に置いた。ノートパソコンの上で、ボールペンで、空いた一行に書いた。

瀬尾さんは、前を見て運転していた。俺が何を書いているのか、のぞきもしなかった。書き終えてカードをしまうとき、一言だけ言った。

「明日、取引先の試験、九時からな」

翌日、受け入れ試験は再開した。その日、AI の確認は七回出た。俺は、七回とも読んだ。そのうち一回は、rm -rf のあとに、思っていたのと違うフォルダの名前が書いてあった。「いいえ」を押した。読めたのはその日の一例で、明日も読めるという約束ではない。

放鷹カード

  • 合図: 許可リスト。通すもの・聞くもの・止めるものを、先に書いておく。聞くのは、決めていないものだけ
  • 検め方: 昨日のログに当てて、確認が何回残るかと、一日一回のものがどこに入るかを数える
  • 止めないもの: 毎日の作業（テスト・lint・型の確認・ビルド・add・commit）を、全許可のときと同じく、確認なしで通すこと。allow に書いていない正しい作業は、止めずに聞く
  • 鷹のせいじゃなかった: 全部聞くか、全部通すかしか持っていなかった。三百十二回のうちの一回を、読まずに押していた
  • 自分の言葉で: 減らしたのは、読むため。名前で止めきれない分は、つながる先で断つ。瀬尾さんの赤も、俺の赤も、どっちも本物だった
