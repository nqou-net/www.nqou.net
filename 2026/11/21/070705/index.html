// Merge は、全部の局の全部の行を、時刻で並べる。同じ時刻なら入力の順。
func Merge(logs []Log) []Entry {
	var table []Entry
	for _, l := range logs {
		for i, r := range l.Records {
			table = append(table, Entry{r, i})
		}
	}
	sort.SliceStable(table, func(i, j int) bool {
		return table[i].At.Before(table[j].At)
	})
	return table
}

全部の局の全部の行を集めて、時刻で並べる。時刻が同じなら、入力の順のまま。ほかのことは、何もしていない。時計が合っているなら、これで起きた順になる。四年間、そう考えて使ってきた。

// estimateOffsets は、送受信の対（同じ通番）から、局ごとに一つの時計のずれを求める。
// 受信が送信より前に書かれている対があれば、送った側の時計が進んでいたとみなす。
func estimateOffsets(logs []timeline.Log) map[string]time.Duration {
	sends := map[string]timeline.Record{}
	for _, l := range logs {
		for _, r := range l.Records {
			if r.Kind == timeline.Send {
				sends[r.Seq] = r
			}
		}
	}
	offsets := map[string]time.Duration{}
	for _, l := range logs {
		for _, r := range l.Records {
			s, ok := sends[r.Seq]
			if r.Kind != timeline.Recv || !ok || r.At.After(s.At) {
				continue
			}
			// 受信が送信より後になる最小の量（1ミリ秒だけ後）
			if d := s.At.Sub(r.At) + time.Millisecond; d > offsets[s.Station] {
				offsets[s.Station] = d
			}
		}
	}
	return offsets
}

通番で、送信と受信の対を作る。受信の時刻が、送信の時刻より前になっている対を探す。その差に一ミリ秒を足した量を、送った側の局のずれとする。一つの局について、いちばん大きい量を一つ。そのずれは、一日を通して一定だと置いた。それを、その局の全部の行から引く関数も、隣に書いた。

func build(logs []Log) *graph {
	g := &graph{index: map[key]int{}}
	sends := map[string][]int{}
	recvs := map[string][]int{}
	for _, l := range logs {
		for i, r := range l.Records {
			n := len(g.table)
			g.table = append(g.table, Entry{r, i})
			g.next = append(g.next, nil)
			g.index[key{r.Station, i}] = n
			if i > 0 { // 規則1: 同じ局のログの中で、前の行は後の行より先
				g.next[n-1] = append(g.next[n-1], n)
			}
			switch r.Kind {
			case Send:
				sends[r.Seq] = append(sends[r.Seq], n)
			case Recv:
				recvs[r.Seq] = append(recvs[r.Seq], n)
			}
		}
	}
	for seq, s := range sends { // 規則2: 送信は、同じ通番の受信より先
		// 送信が一つに決まらない通番は、対を推測で作らない。
		if len(s) != 1 || seq == "" {
			continue
		}
		g.next[s[0]] = append(g.next[s[0]], recvs[seq]...)
	}
	return g
}

func before(g *graph, a, b int) bool {
	seen := make([]bool, len(g.table))
	stack := []int{a}
	for len(stack) > 0 {
		x := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		for _, y := range g.next[x] {
			if y == b {
				return true
			}
			if !seen[y] {
				seen[y] = true
				stack = append(stack, y)
			}
		}
	}
	return false
}

func Merge(logs []Log) []Entry {
	g := build(logs)
	n := len(g.table)
	wait := make([]int, n) // 先の行のうち、まだ並べていないものの数
	for _, outs := range g.next {
		for _, b := range outs {
			wait[b]++
		}
	}
	done := make([]bool, n)
	table := make([]Entry, 0, n)
	for len(table) < n {
		best := -1
		// 並べられる行を探す。記録が矛盾して辺が輪になり、一つも無いときは、行を落とさず時刻だけで取る。
		for _, onlyReady := range []bool{true, false} {
			for i := range g.table {
				if done[i] || (onlyReady && wait[i] > 0) {
					continue
				}
				if best < 0 || g.table[i].At.Before(g.table[best].At) {
					best = i
				}
			}
			if best >= 0 {
				break
			}
		}
		done[best] = true
		table = append(table, g.table[best])
		for _, b := range g.next[best] {
			wait[b]--
		}
	}
	return table
}

並べられる行とは、その行より先に起きたと言える行が、もう全部、表に並んだ行のことだ。wait が、まだ並んでいない先の行の数を数えている。並べられる行の中から、時刻がいちばん早い一行を取る。取ったら、その行から辺の出ている行の wait を一つ減らす。それを、行が尽きるまで繰り返す。

系の出来事の上の関係「→」は、次の三つの条件を満たす、いちばん小さい関係である。一つ。a と b が同じプロセスの出来事で、a が b より前に来るなら、a → b。二つ。a があるプロセスによる電文の送信で、b が別のプロセスによる同じ電文の受信なら、a → b。三つ。a → b で b → c なら、a → c。

// buildStations は、本当の出来事の順（シナリオ）から、局ごとのログを組み立てる。
//   - 行の並び: その局で起きた順（本当の時刻の順）
//   - 行の時刻: 基準 + 本当の時刻 + 書き込みの遅れ + その局の時計のずれ
//
// 時計（clocks）と遅れ（ev.Delay）は注入する。乱数は使わない。
func buildStations(base time.Time, clocks map[string]time.Duration, events []ev) []before.Log {
	events = append([]ev(nil), events...)
	sort.SliceStable(events, func(i, j int) bool { return events[i].At < events[j].At })
	var logs []before.Log
	pos := map[string]int{}
	for _, e := range events {
		i, ok := pos[e.Station]
		if !ok {
			i = len(logs)
			pos[e.Station] = i
			logs = append(logs, before.Log{Station: e.Station})
		}
		logs[i].Records = append(logs[i].Records, before.Record{
			Station: e.Station, At: base.Add(e.At + e.Delay + clocks[e.Station]),
			Kind: e.Kind, Seq: e.Seq, InReplyTo: e.ReplyTo, Summary: e.Summary,
		})
	}
	return logs
}

  • 船について: （記録）船が気象警報の受信を確認した通信は、針路が変わり始めたときより後に起きています。公式報告の時系列表は、各受信局の受信時刻の順に並んでいます
